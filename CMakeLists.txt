cmake_minimum_required(VERSION 3.21)
project(SudokuSAT VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)

add_library(sudoku_core src/sat.c src/dimacs.c src/sudoku.c)
target_include_directories(sudoku_core PUBLIC include)
if(MINGW)
    target_compile_definitions(sudoku_core PUBLIC __USE_MINGW_ANSI_STDIO=1)
endif()
if(MSVC)
    target_compile_definitions(sudoku_core PUBLIC _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(sudoku src/main.c)
target_link_libraries(sudoku PRIVATE sudoku_core)

function(project_checks target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
        if(WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
    if(ENABLE_SANITIZERS)
        if(MSVC)
            message(FATAL_ERROR "ENABLE_SANITIZERS requires GCC or Clang")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

project_checks(sudoku_core)
project_checks(sudoku)
include(CTest)
if(BUILD_TESTING)
    add_executable(sudoku_tests tests/test_core.c)
    target_link_libraries(sudoku_tests PRIVATE sudoku_core)
    project_checks(sudoku_tests)
    foreach(suite solver dimacs sudoku generator)
        add_test(NAME ${suite} COMMAND sudoku_tests ${suite} ${CMAKE_CURRENT_SOURCE_DIR}/tests/data)
        set_tests_properties(${suite} PROPERTIES TIMEOUT 60)
    endforeach()
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli_and_differential
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py
                $<TARGET_FILE:sudoku> ${CMAKE_CURRENT_SOURCE_DIR})
        set_tests_properties(cli_and_differential PROPERTIES TIMEOUT 120)
    endif()
endif()

install(TARGETS sudoku RUNTIME DESTINATION bin)
